Autodarts Kiosk APT Repository

curl -fsSL https://beyond-agentur.github.io/autodarts-kiosk-apt/autodarts-kiosk-archive-keyring.asc \
  | sudo tee /usr/share/keyrings/autodarts-kiosk-archive-keyring.asc >/dev/null

echo "deb [signed-by=/usr/share/keyrings/autodarts-kiosk-archive-keyring.asc] https://beyond-agentur.github.io/autodarts-kiosk-apt stable main" \
  | sudo tee /etc/apt/sources.list.d/autodarts-kiosk.list >/dev/null

sudo apt update
sudo apt install autodarts-kiosk
